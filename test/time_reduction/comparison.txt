report for distance
to-report matrix-dist [origin dest matrix]
  let header item 0 matrix
  let col position  [pro_com] of dest  header
  if col = false [ report nobody ]

  let rows but-first matrix
  let row first filter [r -> first r = [pro_com] of origin] rows
  if row = nobody [ report nobody ]

  report item col row
end
